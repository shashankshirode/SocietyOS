import React from 'react';
import { StyleSheet, Text, View } from 'react-native';
import { Colors } from '../../../shared/constants/colors';
import { Spacing } from '../../../shared/constants/spacing';
import { Typography } from '../../../shared/constants/typography';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { StatusBadge, getMeetingStatusBadgeType } from '../../../shared/components/StatusBadge';
import { ActionTile, DetailCard, ParkingScreen } from '../../parking/components/ParkingUi';
import { useMeetingDetail } from '../data/useMeetingDetail';
import type { MeetingDetailScreenProps } from '../../../app/navigation/navigation.types';
 
export function MeetingDetailScreen({ navigation, route }: MeetingDetailScreenProps) {
  const { meetingId } = route.params;
  const { data: meeting, isLoading, error, refetch } = useMeetingDetail(meetingId);
 
  if (isLoading) return <LoadingState message="Loading meeting details..." showCardPlaceholder />;
  if (error) return <ErrorState message={error.message} onRetry={refetch} />;
  if (!meeting) return <ErrorState message="Meeting not found." onRetry={refetch} />;
 
  const quorumPercent = meeting.totalMembers > 0 ? Math.round(((meeting.presentCount + meeting.proxyCount) / meeting.totalMembers) * 100) : 0;
 
  return (
    <ParkingScreen title={meeting.meetingType} subtitle={meeting.title} onBack={navigation.goBack}>
      <View style={styles.statusRow}>
        <StatusBadge label={meeting.status.replace(/_/g, ' ')} type={getMeetingStatusBadgeType(meeting.status)} />
        <View style={styles.typeChip}>
          <Text style={styles.typeText}>{meeting.meetingType}</Text>
        </View>
      </View>
 
      <DetailCard title="Meeting Information">
        <View style={styles.infoRow}><Text style={styles.infoLabel}>Date</Text><Text style={styles.infoValue}>{meeting.scheduledDate}</Text></View>
        <View style={styles.infoRow}><Text style={styles.infoLabel}>Time</Text><Text style={styles.infoValue}>{meeting.scheduledTime}{meeting.endTime ? ` – ${meeting.endTime}` : ''}</Text></View>
        <View style={styles.infoRow}><Text style={styles.infoLabel}>Venue</Text><Text style={styles.infoValue}>{meeting.venue}</Text></View>
        <View style={styles.infoRow}><Text style={styles.infoLabel}>Organizer</Text><Text style={styles.infoValue}>{meeting.organizer}</Text></View>
        {meeting.noticeDate && <View style={styles.infoRow}><Text style={styles.infoLabel}>Notice Date</Text><Text style={styles.infoValue}>{meeting.noticeDate}</Text></View>}
      </DetailCard>
 
      <DetailCard title="Quorum Status">
        <View style={styles.quorumBar}>
          <View style={[styles.quorumFill, { width: `${Math.min(quorumPercent, 100)}%` }]} />
        </View>
        <Text style={styles.quorumText}>{quorumPercent}% attendance ({meeting.presentCount + meeting.proxyCount}/{meeting.totalMembers}) · Required: {meeting.quorumRequired}%</Text>
        <View style={styles.attendanceSummary}>
          <Text style={styles.attendanceItem}>✓ Confirmed: {meeting.confirmedCount}</Text>
          <Text style={styles.attendanceItem}>👤 Present: {meeting.presentCount}</Text>
          <Text style={styles.attendanceItem}>📋 Proxy: {meeting.proxyCount}</Text>
        </View>
      </DetailCard>
 
      <Text style={styles.descriptionTitle}>Description</Text>
      <Text style={styles.description}>{meeting.description}</Text>
 
      <DetailCard title="Quick Actions">
        <ActionTile title="View Agenda" iconName="list-outline" onPress={() => navigation.navigate('MeetingAgenda', { meetingId })} />
        <ActionTile title="Attendance" iconName="people-outline" onPress={() => navigation.navigate('MeetingAttendance', { meetingId })} />
        {meeting.status === 'SCHEDULED' && <ActionTile title="RSVP" iconName="checkmark-circle-outline" onPress={() => navigation.navigate('MeetingRsvp', { meetingId })} />}
        {meeting.resolutionCount > 0 && <ActionTile title="Resolutions" iconName="document-text-outline" onPress={() => navigation.navigate('ResolutionList')} />}
        {meeting.hasMinutes && <ActionTile title="Meeting Minutes" iconName="reader-outline" onPress={() => navigation.navigate('MeetingMinutesDetail', { meetingId })} />}
      </DetailCard>
    </ParkingScreen>
  );
}
 
const styles = StyleSheet.create({
  statusRow: { flexDirection: 'row', alignItems: 'center', gap: Spacing.sm, marginBottom: Spacing.md },
  typeChip: { backgroundColor: Colors.primaryLight, paddingHorizontal: Spacing.sm, paddingVertical: 2, borderRadius: 6 },
  typeText: { ...Typography.caption, color: Colors.primary, fontWeight: '700' },
  infoRow: { flexDirection: 'row', justifyContent: 'space-between', paddingVertical: Spacing.xs, borderBottomWidth: StyleSheet.hairlineWidth, borderBottomColor: Colors.border },
  infoLabel: { ...Typography.bodySmall, color: Colors.textTertiary },
  infoValue: { ...Typography.bodySmall, color: Colors.textPrimary, fontWeight: '500', maxWidth: '60%', textAlign: 'right' },
  quorumBar: { height: 8, backgroundColor: Colors.neutralLight, borderRadius: 4, overflow: 'hidden', marginBottom: Spacing.sm },
  quorumFill: { height: '100%', backgroundColor: Colors.success, borderRadius: 4 },
  quorumText: { ...Typography.caption, color: Colors.textSecondary, marginBottom: Spacing.sm },
  attendanceSummary: { flexDirection: 'row', gap: Spacing.lg },
  attendanceItem: { ...Typography.caption, color: Colors.textSecondary },
  descriptionTitle: { ...Typography.bodyLarge, color: Colors.textPrimary, fontWeight: '600', marginBottom: Spacing.sm, marginTop: Spacing.md },
  description: { ...Typography.bodySmall, color: Colors.textSecondary, lineHeight: 20, marginBottom: Spacing.md },
});
