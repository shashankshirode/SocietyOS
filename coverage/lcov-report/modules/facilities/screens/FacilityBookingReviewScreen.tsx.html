import React from 'react';
import type { NativeStackScreenProps } from '@react-navigation/native-stack';
import { AppButton } from '../../../shared/components/AppButton';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import type { FacilityStackParamList } from '../../../app/navigation/navigation.types';
import { DetailCard, DetailRow, ParkingScreen, WarningText } from '../../parking/components/ParkingUi';
import { useFacilityBookingReview } from '../data/useFacilityBookingReview';
 
type Props = NativeStackScreenProps<FacilityStackParamList, 'FacilityBookingReview'>;
 
export function FacilityBookingReviewScreen({ navigation, route }: Props) {
  const bookingId = route.params.bookingId ?? route.params.draftBookingId ?? 'facility-booking-1';
  const { data: booking, isLoading, error, refetch } = useFacilityBookingReview(bookingId);
  if (isLoading) return <LoadingState message="Loading booking review..." showCardPlaceholder />;
  if (error) return <ErrorState message={error.message} onRetry={refetch} />;
  if (!booking) return <ErrorState message="Booking review is unavailable." onRetry={refetch} />;
  return (
    <ParkingScreen title="Review Booking" subtitle={booking.bookingNumber} onBack={navigation.goBack}>
      <DetailCard title="Booking summary">
        <DetailRow label="Facility" value={booking.facilityName} />
        <DetailRow label="Date" value={booking.date} />
        <DetailRow label="Slot" value={booking.slot} />
        <DetailRow label="Purpose" value={booking.purpose} />
        <DetailRow label="Guests" value={`${booking.guestCount}`} />
        <DetailRow label="Charges" value={`Rs. ${booking.chargeAmount}`} />
        <DetailRow label="Deposit" value={`Rs. ${booking.depositAmount}`} />
        <DetailRow label="Total payable placeholder" value={`Rs. ${booking.chargeAmount + booking.depositAmount}`} />
        <DetailRow label="Approval" value={booking.approvalStatus} />
        <DetailRow label="Cancellation rule" value={booking.cancellationEligibility} isLast />
      </DetailCard>
      <WarningText>Payment/deposit integration will be connected later. This mock booking is created without real payment.</WarningText>
      <AppButton title="Confirm Booking" onPress={() => navigation.replace('FacilityBookingConfirmation', { bookingId: booking.id })} />
    </ParkingScreen>
  );
}
