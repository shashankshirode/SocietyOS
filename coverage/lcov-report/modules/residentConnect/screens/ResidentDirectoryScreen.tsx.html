import React, { useState, useMemo } from 'react';
import { FlatList, Pressable, StyleSheet, Text, TextInput, View, ScrollView } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import Ionicons from '@expo/vector-icons/Ionicons';
import Animated, { FadeInLeft } from 'react-native-reanimated';
import { Colors } from '../../../shared/constants/colors';
import { Layout } from '../../../shared/constants/layout';
import { Spacing } from '../../../shared/constants/spacing';
import { Typography } from '../../../shared/constants/typography';
import { AppCard } from '../../../shared/cards/AppCard';
import { AppHeader } from '../../../shared/components/AppHeader';
import { EmptyState } from '../../../shared/feedback/EmptyState';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import {
  StatusBadge,
  getDirectoryVisibilityBadgeType,
  getResidentConnectionBadgeType,
} from '../../../shared/components/StatusBadge';
import { useResidentDirectory } from '../data/useResidentDirectory';
import type { ResidentDirectoryScreenProps } from '../../../app/navigation/navigation.types';
import type { ResidentDirectoryEntry } from '../../../shared/types/residentConnect.types';
 
export function ResidentDirectoryScreen({ navigation }: ResidentDirectoryScreenProps) {
  const [search, setSearch] = useState('');
  const [towerFilter, setTowerFilter] = useState<'ALL' | 'A Wing' | 'B Wing' | 'C Wing'>('ALL');
  const [typeFilter, setTypeFilter] = useState<'ALL' | 'OWNER' | 'TENANT'>('ALL');
 
  // Directory hooks
  const { data: directory = [], isLoading, error, refetch } = useResidentDirectory();
 
  const filteredDirectory = useMemo(() => {
    return directory.filter((item) => {
      // Search match
      const query = search.toLowerCase().trim();
      const nameMatch = item.name.toLowerCase().includes(query);
      const flatMatch = item.flatNumber.toLowerCase().includes(query);
      const searchMatch = !query || nameMatch || flatMatch;
 
      // Tower filter
      const towerMatch = towerFilter === 'ALL' || item.tower === towerFilter;
 
      // Type filter
      const typeMatch = typeFilter === 'ALL' || item.residentType === typeFilter;
 
      return searchMatch && towerMatch && typeMatch;
    });
  }, [directory, search, towerFilter, typeFilter]);
 
  if (isLoading) return <LoadingState />;
  if (error) {
    return (
      <ErrorState
        title="Failed to Load Directory"
        message="Could not retrieve society directory index."
        onRetry={refetch}
      />
    );
  }
 
  const renderDirectoryItem = ({ item, index }: { item: ResidentDirectoryEntry; index: number }) => {
    const isHidden = item.visibilityStatus === 'HIDDEN';
    const isLimited = item.visibilityStatus === 'LIMITED';
 
    return (
      <Animated.View entering={FadeInLeft.delay(index * 30).duration(350)}>
        <AppCard style={styles.card}>
          <View style={styles.cardHeader}>
            <View>
              <Text style={styles.flatText}>{item.tower} · Flat {item.flatNumber}</Text>
              <Text style={styles.residentTypeLabel}>{item.residentType}</Text>
            </View>
            <View style={styles.badgesRow}>
              <StatusBadge
                label={item.visibilityStatus}
                type={getDirectoryVisibilityBadgeType(item.visibilityStatus)}
                style={styles.badge}
              />
              <StatusBadge
                label={item.connectionStatus.replace(/_/g, ' ')}
                type={getResidentConnectionBadgeType(item.connectionStatus)}
                style={styles.badge}
              />
            </View>
          </View>
 
          <View style={styles.divider} />
 
          {isHidden ? (
            <Text style={styles.optOutText}>
              <Ionicons name="eye-off-outline" size={14} color={Colors.textMuted} /> This resident has limited their directory visibility.
            </Text>
          ) : (
            <View style={styles.profileDetails}>
              <Text style={styles.nameText}>
                {isLimited ? `${item.name.charAt(0)}***` : item.name}
              </Text>
              {item.bio && <Text style={styles.bioText} numberOfLines={1}>{item.bio}</Text>}
 
              {item.mutualContext && item.mutualContext.length > 0 && (
                <View style={styles.contextTagsRow}>
                  {item.mutualContext.map((ctx) => (
                    <View key={ctx} style={styles.contextTag}>
                      <Text style={styles.contextTagText}>{ctx}</Text>
                    </View>
                  ))}
                </View>
              )}
            </View>
          )}
 
          <Pressable
            style={({ pressed }) => [styles.viewBtn, pressed && styles.pressed]}
            onPress={() => navigation.navigate('ResidentPreview', { residentId: item.id })}
          >
            <Text style={styles.viewBtnText}>View Preview</Text>
            <Ionicons name="arrow-forward" size={14} color={Colors.primary} />
          </Pressable>
        </AppCard>
      </Animated.View>
    );
  };
 
  return (
    <SafeAreaView style={styles.safeArea} edges={['top']}>
      <AppHeader
        title="Society Directory"
        showBack
        onBack={() => navigation.goBack()}
      />
      <View style={styles.mainContainer}>
        {/* Search */}
        <View style={styles.searchBarContainer}>
          <Ionicons name="search-outline" size={18} color={Colors.textMuted} style={styles.searchIcon} />
          <TextInput
            value={search}
            onChangeText={setSearch}
            placeholder="Search by name, flat number..."
            placeholderTextColor={Colors.textMuted}
            style={styles.searchInput}
          />
        </View>
 
        {/* Filters Scroll */}
        <View style={styles.filterScrollOuter}>
          <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.filterScroll}>
            {(['ALL', 'A Wing', 'B Wing', 'C Wing'] as const).map((wing) => (
              <Pressable
                key={wing}
                style={[styles.filterChip, towerFilter === wing && styles.filterChipActive]}
                onPress={() => setTowerFilter(wing)}
              >
                <Text style={[styles.filterChipText, towerFilter === wing && styles.filterChipTextActive]}>
                  {wing === 'ALL' ? 'All Wings' : wing}
                </Text>
              </Pressable>
            ))}
          </ScrollView>
        </View>
 
        <View style={[styles.filterScrollOuter, { marginTop: 4, marginBottom: Spacing.sm }]}>
          <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.filterScroll}>
            {(['ALL', 'OWNER', 'TENANT'] as const).map((type) => (
              <Pressable
                key={type}
                style={[styles.filterChip, typeFilter === type && styles.filterChipActive]}
                onPress={() => setTypeFilter(type)}
              >
                <Text style={[styles.filterChipText, typeFilter === type && styles.filterChipTextActive]}>
                  {type === 'ALL' ? 'All Types' : type}
                </Text>
              </Pressable>
            ))}
          </ScrollView>
        </View>
 
        <FlatList
          data={filteredDirectory}
          renderItem={renderDirectoryItem}
          keyExtractor={(item) => item.id}
          contentContainerStyle={styles.listContent}
          showsVerticalScrollIndicator={false}
          ListEmptyComponent={
            <EmptyState
              title="No Directory Logs"
              description="No matching society residents found."
              iconName="people-outline"
            />
          }
        />
      </View>
    </SafeAreaView>
  );
}
