// 3. HousekeepingRoundDetailScreen
export function HousekeepingRoundDetailScreen({ route, navigation }: any) {
  const { roundId } = route.params;
  const { data, refetch } = useHousekeepingRoundDetail(roundId);
 
  const startMutation = useStartHousekeepingRound();
  const completeMutation = useCompleteHousekeepingRound();
 
  if (!data) {
    return (
      <ScreenContainer edges={['top', 'bottom']}>
        <ResponsivePageHeader title="Round Detail" onBack={() => navigation.goBack()} />
        <View style={styles.center}>
          <Text style={styles.errorText}>Loading round details...</Text>
        </View>
      </ScreenContainer>
    );
  }
 
  const round = data as any;
 
  const handleStart = async () => {
    await startMutation.mutateAsync(roundId);
    Alert.alert('Success', 'Housekeeping round started.');
    refetch();
  };
 
  const handleComplete = () => {
    Alert.alert(
      'Complete Round',
      'Are you sure you want to mark this round as completed?',
      [
        { text: 'Cancel', style: 'cancel' },
        { 
          text: 'Complete', 
          onPress: async () => {
            await completeMutation.mutateAsync({
              roundId,
              input: { completedBy: 'Housekeeping Staff', remarks: 'Completed SOP cleaning' }
            });
            Alert.alert('Success', 'Housekeeping round completed.');
            refetch();
          } 
        }
      ]
    );
  };
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader 
        title={round.roundNumber} 
        subtitle={round.area} 
        onBack={() => navigation.goBack()} 
      />
      <ScrollView contentContainerStyle={styles.scroll} showsVerticalScrollIndicator={false}>
        <AppCard padding="lg" style={{ marginBottom: Spacing.lg }}>
          <InfoRow label="Area" value={round.area} />
          <InfoRow label="Cleaning Type" value={round.cleaningType.replace(/_/g, ' ')} />
          <InfoRow label="Frequency" value={round.frequency} />
          <InfoRow label="Status" value={round.status} />
          <InfoRow label="Assigned Staff" value={round.assignedStaff} />
          <InfoRow label="Scheduled Time" value={round.scheduledTime} isLast />
        </AppCard>
 
        <View style={styles.actionSection}>
          {round.status === 'SCHEDULED' && (
            <AppButton title="Start Cleaning Round" onPress={handleStart} variant="primary" fullWidth />
          )}
          {round.status === 'IN_PROGRESS' && (
            <View style={styles.btnRow}>
              <AppButton 
                title="SOP Checklist" 
                onPress={() => navigation.navigate('FloorCleaningChecklist', { roundId })} 
                variant="outline" 
                style={{ flex: 1, marginRight: Spacing.sm }} 
              />
              <AppButton 
                title="Complete Round" 
                onPress={handleComplete} 
                variant="success" 
                style={{ flex: 1 }} 
              />
            </View>
          )}
          {round.status === 'COMPLETED' && (
            <AppButton 
              title="Verify Round" 
              onPress={() => navigation.navigate('HousekeepingSupervisorVerification', { roundId })} 
              variant="primary" 
              fullWidth 
            />
          )}
        </View>
 
        <ComplianceSafetyNotice />
      </ScrollView>
    </ScreenContainer>
  );
}
 
// 4. FloorCleaningChecklistScreen
export function FloorCleaningChecklistScreen({ route, navigation }: any) {
  const { roundId } = route.params;
  const [swept, setSwept] = useState(false);
  const [mopped, setMopped] = useState(false);
  const [disinfected, setDisinfected] = useState(false);
  const [trashEmptied, setTrashEmptied] = useState(false);
 
  const checklistMutation = useSubmitFloorCleaningChecklist();
 
  const handleSubmit = async () => {
    await checklistMutation.mutateAsync({
      roundId,
      swept,
      mopped,
      disinfected,
      trashEmptied,
    });
    Alert.alert('Success', 'SOP checklist submitted successfully.');
    navigation.goBack();
  };
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader title="Floor Cleaning SOP Checklist" onBack={() => navigation.goBack()} />
      <ScrollView contentContainerStyle={styles.scroll} showsVerticalScrollIndicator={false}>
        <AppCard padding="lg" style={{ marginBottom: Spacing.xl }}>
          <Text style={styles.checklistHeader}>Verify the following cleaning steps:</Text>
          
          <InspectionChecklistCard title="Area Sweeping Completed" checked={swept} onPress={() => setSwept(!swept)} />
          <InspectionChecklistCard title="Wet Mopping with Disinfectant" checked={mopped} onPress={() => setMopped(!mopped)} />
          <InspectionChecklistCard title="High-touch Surfaces Disinfected" checked={disinfected} onPress={() => setDisinfected(!disinfected)} />
          <InspectionChecklistCard title="Trash Bins Emptied and Re-lined" checked={trashEmptied} onPress={() => setTrashEmptied(!trashEmptied)} />
 
          <AppButton 
            title={checklistMutation.isPending ? 'Saving...' : 'Save Checklist'} 
            onPress={handleSubmit} 
            variant="primary" 
            fullWidth 
            disabled={checklistMutation.isPending}
            style={{ marginTop: Spacing.lg }}
          />
        </AppCard>
      </ScrollView>
    </ScreenContainer>
  );
}
 
// 5. CommonAreaInspectionScreen
export function CommonAreaInspectionScreen({ navigation }: any) {
  const [area, setArea] = useState('');
  const [cleanlinessRating, setCleanlinessRating] = useState('4');
  const [odorRating, setOdorRating] = useState('4');
  const [issuesObserved, setIssuesObserved] = useState('');
 
  const inspectionMutation = useSubmitCommonAreaInspection();
 
  const handleSubmit = async () => {
    if (!area) {
      Alert.alert('Error', 'Please enter the area inspected.');
      return;
    }
    await inspectionMutation.mutateAsync({
      area,
      cleanlinessRating: parseInt(cleanlinessRating),
      odorRating: parseInt(odorRating),
      issuesObserved,
    });
    Alert.alert('Success', 'Inspection report logged.');
    navigation.goBack();
  };
 
  const ratingOptions = [
    { label: '1 (Poor)', value: '1' },
    { label: '2', value: '2' },
    { label: '3 (Fair)', value: '3' },
    { label: '4', value: '4' },
    { label: '5 (Excellent)', value: '5' },
  ];
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader title="Common Area Inspection" onBack={() => navigation.goBack()} />
      <ScrollView contentContainerStyle={styles.scroll} showsVerticalScrollIndicator={false}>
        <AppCard padding="lg" style={{ marginBottom: Spacing.xl }}>
          <FormField 
            label="Inspected Area" 
            value={area} 
            onChangeText={setArea} 
            placeholder="e.g., Clubhouse Lobby" 
            required 
          />
 
          <Text style={styles.formLabel}>Cleanliness Rating *</Text>

            selected={cleanlinessRating}

          <FilterChips 
            options={ratingOptions}

            label="Issues Observed"

            title={inspectionMutation.isPending ? 'Submitting...' : 'Submit Inspection'}

            fullWidth 
            disabled={inspectionMutation.isPending}
            style={{ marginTop: Spacing.md }}
          />
        </AppCard>
      </ScrollView>

// 7. HousekeepingIssueReportScreen
