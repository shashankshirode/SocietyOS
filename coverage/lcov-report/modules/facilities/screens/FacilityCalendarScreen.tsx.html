import React from 'react';
import { FlatList } from 'react-native';
import type { NativeStackScreenProps } from '@react-navigation/native-stack';
import { AppCard } from '../../../shared/cards/AppCard';
import { FilterChips } from '../../../shared/lists/FilterChips';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import type { FacilityStackParamList } from '../../../app/navigation/navigation.types';
import { DetailRow, ParkingScreen } from '../../parking/components/ParkingUi';
import { useFacilityCalendar } from '../data/useFacilityCalendar';
 
type Props = NativeStackScreenProps<FacilityStackParamList, 'FacilityCalendar'>;
type DateKey = '2026-07-01' | '2026-07-02' | '2026-07-03';
const dates = [{ key: '2026-07-01', label: 'Jul 1' }, { key: '2026-07-02', label: 'Jul 2' }, { key: '2026-07-03', label: 'Jul 3' }] satisfies Array<{ key: DateKey; label: string }>;
 
export function FacilityCalendarScreen({ navigation }: Props) {
  const [date, setDate] = React.useState<DateKey>('2026-07-01');
  const { data = [], isLoading, error, refetch } = useFacilityCalendar(date);
  if (isLoading) return <LoadingState message="Loading calendar..." showCardPlaceholder />;
  if (error) return <ErrorState message={error.message} onRetry={refetch} />;
  return (
    <ParkingScreen title="Booking Calendar" subtitle="Simple mock calendar view" onBack={navigation.goBack}>
      <FilterChips options={dates} selectedKey={date} onSelect={setDate} />
      <FlatList data={data} keyExtractor={(item) => item.facilityName} scrollEnabled={false} renderItem={({ item }) => (
        <AppCard style={{ marginBottom: 12 }}>
          <DetailRow label="Facility" value={item.facilityName} />
          <DetailRow label="Bookings" value={`${item.bookingCount}`} />
          <DetailRow label="Available slots" value={`${item.availableSlotCount}`} />
          <DetailRow label="My bookings" value={`${item.userBookingCount}`} />
          <DetailRow label="Fully booked" value={item.isFullyBooked ? 'Yes' : 'No'} />
          <DetailRow label="Maintenance blocked" value={item.hasMaintenanceBlock ? 'Yes' : 'No'} isLast />
        </AppCard>
      )} />
    </ParkingScreen>
  );
}
