import React from 'react';
import type { NativeStackScreenProps } from '@react-navigation/native-stack';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { FormField } from '../../../shared/forms/FormField';
import type { FacilityStackParamList } from '../../../app/navigation/navigation.types';
import { FooterActions, MockFilePicker, ParkingScreen, Selector } from '../../parking/components/ParkingUi';
import { useGuestRoomBooking } from '../data/useGuestRoomBooking';
 
type Props = NativeStackScreenProps<FacilityStackParamList, 'GuestRoomBooking'>;
 
export function GuestRoomBookingScreen({ navigation }: Props) {
  const [guestName, setGuestName] = React.useState('');
  const [guestMobile, setGuestMobile] = React.useState('');
  const [checkInDate, setCheckInDate] = React.useState('');
  const [checkOutDate, setCheckOutDate] = React.useState('');
  const [guestCount, setGuestCount] = React.useState('');
  const [purpose, setPurpose] = React.useState('');
  const [idProofLabel, setIdProofLabel] = React.useState('');
  const [requirements, setRequirements] = React.useState('');
  const [rulesAccepted, setRulesAccepted] = React.useState<'NO' | 'YES'>('NO');
  const [errors, setErrors] = React.useState<Record<string, string>>({});
  const { submit, isSubmitting, error } = useGuestRoomBooking();
  async function handleSubmit() {
    const next: Record<string, string> = {};
    const guests = Number(guestCount);
    if (!guestName.trim()) next.guestName = 'Guest name is required.';
    if (!/^\d{10}$/.test(guestMobile.trim())) next.guestMobile = 'Guest mobile must be 10 digits.';
    if (!checkInDate.trim()) next.checkInDate = 'Check-in date is required.';
    if (!checkOutDate.trim()) next.checkOutDate = 'Check-out date is required.';
    if (checkInDate && checkOutDate && checkOutDate <= checkInDate) next.checkOutDate = 'Check-out date must be after check-in date.';
    if (!guestCount.trim() || guests <= 0) next.guestCount = 'Number of guests is required.';
    if (!purpose.trim()) next.purpose = 'Purpose is required.';
    if (rulesAccepted !== 'YES') next.rulesAccepted = 'Guest room rules agreement is required.';
    setErrors(next);
    if (Object.keys(next).length) return;
    const result = await submit({ guestName, guestMobile, checkInDate, checkOutDate, guestCount: guests, purpose, idProofLabel, specialRequirements: requirements, rulesAccepted: true });
    if (result.ok) navigation.navigate('FacilityBookingReview', { bookingId: result.data.id });
  }
  return (
    <ParkingScreen title="Guest Room Booking" subtitle="Mock overnight booking" onBack={navigation.goBack} footer={<FooterActions primaryLabel="Review guest room booking" onPrimary={handleSubmit} loading={isSubmitting} />}>
      {error ? <ErrorState message={error.message} /> : null}
      <FormField label="Guest name" required value={guestName} onChangeText={setGuestName} error={errors.guestName} />
      <FormField label="Guest mobile" required value={guestMobile} onChangeText={setGuestMobile} keyboardType="phone-pad" maxLength={10} error={errors.guestMobile} />
      <FormField label="Check-in date" required value={checkInDate} onChangeText={setCheckInDate} error={errors.checkInDate} placeholder="2026-07-01" />
      <FormField label="Check-out date" required value={checkOutDate} onChangeText={setCheckOutDate} error={errors.checkOutDate} placeholder="2026-07-02" />
      <FormField label="Number of guests" required value={guestCount} onChangeText={setGuestCount} keyboardType="number-pad" error={errors.guestCount} />
      <FormField label="Purpose of stay" required value={purpose} onChangeText={setPurpose} error={errors.purpose} />
      <MockFilePicker label="ID proof placeholder" fileName={idProofLabel} helper="Tap to select mock ID proof file" onPress={() => setIdProofLabel('guest-id-proof.pdf')} />
      <FormField label="Special requirements" value={requirements} onChangeText={setRequirements} multiline numberOfLines={3} />
      <Selector label="Guest room rules" value={rulesAccepted} onChange={setRulesAccepted} error={errors.rulesAccepted} options={[{ label: 'I agree', value: 'YES' }, { label: 'Not yet', value: 'NO' }]} />
    </ParkingScreen>
  );
}
