import React from 'react';
import { StyleSheet, View } from 'react-native';
import Animated, { FadeInDown, FadeInUp } from 'react-native-reanimated';
import { useAppTheme } from '../../../shared/theme/useAppTheme';
import { useAuthSession } from '../../../core/auth/useAuthSession';
import { AppModeSelectorScreenProps, RootStackParamList } from '../../../app/navigation/navigation.types';
import { ScreenScaffold } from '../../../shared/layout/ScreenScaffold';
import { Stack } from '../../../shared/layout/Stack';
import { Box } from '../../../shared/layout/Box';
import { AppText } from '../../../shared/components/AppText';
import { AppCard } from '../../../shared/cards/AppCard';
import { AppIconBubble } from '../../../shared/icons/AppIconBubble';
import { ResponsiveGrid } from '../../../shared/layout/ResponsiveGrid';
 
export function AppModeSelectorScreen({ navigation }: AppModeSelectorScreenProps) {
  const { colors, radius } = useAppTheme();
  const { startMockSessionForRoute } = useAuthSession();
 
  const handleSelectMode = async (mode: keyof RootStackParamList) => {
    await startMockSessionForRoute(mode);
    navigation.reset({
      index: 0,
      routes: [{ name: mode }],
    });
  };
 
  const modes: {
    name: keyof RootStackParamList;
    title: string;
    description: string;
    icon: any;
    color: string;
    bgColor: string;
  }[] = [
    {
      name: 'ResidentApp',
      title: 'Resident App',
      description: 'Passes, bills, notices, and helpdesk for society residents.',
      icon: 'home',
      color: colors.primary,
      bgColor: colors.primarySoft,
    },
    {
      name: 'GuardApp',
      title: 'Guard App',
      description: 'Gate entries, expected visitors lists, manual verification, and gate logs.',
      icon: 'guard',
      color: colors.guard || '#0F766E',
      bgColor: `${colors.guard || '#0F766E'}15`,
    },
    {
      name: 'FacilityManagerApp',
      title: 'Facility Manager App',
      description: 'Vendors, AMCs, assets, compliance, and work orders for operations.',
      icon: 'facility',
      color: colors.facility || '#2563EB',
      bgColor: `${colors.facility || '#2563EB'}15`,
    },
    {
      name: 'SocietyAdminApp',
      title: 'Society Admin Console',
      description: 'Manage units, residents, approvals, notices, complaints, and audit logs.',
      icon: 'admin',
      color: colors.admin || '#7C3AED',
      bgColor: `${colors.admin || '#7C3AED'}15`,
    },
    {
      name: 'TreasurerApp',
      title: 'Treasurer Dashboard',
      description: 'Financial collections, charges, bills, defaulter reports, and flat ledgers.',
      icon: 'treasurer',
      color: colors.treasurer || '#B45309',
      bgColor: `${colors.treasurer || '#B45309'}15`,
    },
  ];
 
  return (
    <ScreenScaffold scroll edges={['top', 'bottom']} style={styles.container}>
      <Stack gap="xl">
        <Animated.View entering={FadeInUp.duration(500)} style={styles.header}>
          <AppText variant="displayMedium" style={[styles.brandTitle, { color: colors.primary }]}>
            SocietyOS
          </AppText>
          <AppText variant="screenSubtitle" style={{ color: colors.textSecondary }}>
            Select your portal to continue
          </AppText>
        </Animated.View>
 
        <Animated.View entering={FadeInDown.delay(100).duration(500)}>
          <ResponsiveGrid columnsPhone={1} columnsTablet={2} gap={16}>
            {modes.map((mode) => (
              <AppCard
                key={mode.name}
                pressable
                onPress={() => handleSelectMode(mode.name)}
                style={styles.modeCard}
              >
                <Box alignItems="center" style={{ width: '100%' }}>
                  <AppIconBubble
                    name={mode.icon}
                    size={60}
                    iconSize={28}
                    color={mode.color}
                    backgroundColor={mode.bgColor}
                  />
                  <AppText variant="sectionTitle" style={[styles.modeTitle, { color: colors.textPrimary }]}>
                    {mode.title}
                  </AppText>
                  <AppText variant="bodySmall" style={[styles.modeDesc, { color: colors.textSecondary }]}>
                    {mode.description}
                  </AppText>
                </Box>
              </AppCard>
            ))}
          </ResponsiveGrid>
        </Animated.View>
 
        <Animated.View entering={FadeInDown.delay(200).duration(500)} style={styles.footer}>
          <AppText variant="caption" style={[styles.footerText, { color: colors.textMuted }]}>
            Note: This selector is a dev-only tool simulating role-based routing before auth integration.
            Advanced platform and hardware modules are paused for MVP.
          </AppText>
        </Animated.View>
      </Stack>
    </ScreenScaffold>
  );
}
 
const styles = StyleSheet.create({
  container: {
    flex: 1,
  },
  header: {
    alignItems: 'center',
    marginTop: 24,
  },
  brandTitle: {
    fontWeight: '800',
    letterSpacing: -0.5,
  },
  modeCard: {
    alignItems: 'center',
    paddingVertical: 24,
    width: '100%',
  },
  modeTitle: {
    fontWeight: '700',
    marginTop: 16,
    marginBottom: 8,
  },
  modeDesc: {
    textAlign: 'center',
    paddingHorizontal: 12,
    lineHeight: 18,
  },
  footer: {
    marginTop: 12,
    alignItems: 'center',
    paddingBottom: 24,
  },
  footerText: {
    textAlign: 'center',
    paddingHorizontal: 16,
    lineHeight: 16,
  },
});
