// 4. ComplianceTaskDetailScreen
export function ComplianceTaskDetailScreen({ route, navigation }: any) {
  const { taskId } = route.params;
  const { data, refetch } = useComplianceTaskDetail(taskId);
 
  const startMutation = useStartComplianceTask();
  const completeMutation = useCompleteComplianceTask();
  const verifyMutation = useVerifyComplianceTask();
  const reopenMutation = useReopenComplianceTask();
 
  if (!data) {
    return (
      <ScreenContainer edges={['top', 'bottom']}>
        <ResponsivePageHeader title="Task Detail" onBack={() => navigation.goBack()} />
        <View style={styles.center}>
          <Text style={styles.errorText}>Loading task details...</Text>
        </View>
      </ScreenContainer>
    );
  }
 
  const task = data as any;
 
  const handleStart = async () => {
    await startMutation.mutateAsync(taskId);
    Alert.alert('Success', 'Task started successfully.');
    refetch();
  };
 
  const handleComplete = async () => {
    await completeMutation.mutateAsync({ taskId, input: { notes: 'Completed as per SOP.' } });
    Alert.alert('Success', 'Task marked as completed.');
    refetch();
  };
 
  const handleVerify = async () => {
    await verifyMutation.mutateAsync({ taskId, input: { notes: 'Inspection verified and found satisfactory.' } });
    Alert.alert('Success', 'Task verified successfully.');
    refetch();
  };
 
  const handleReopen = async () => {
    await reopenMutation.mutateAsync({ taskId, input: { notes: 'Needs rework. Reopening.' } });
    Alert.alert('Success', 'Task reopened.');
    refetch();
  };
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader 
        title={task.taskNumber} 
        subtitle={task.title} 
        onBack={() => navigation.goBack()} 
      />
      <ScrollView contentContainerStyle={styles.scroll} showsVerticalScrollIndicator={false}>
        <AppCard style={styles.detailCard} padding="lg">
          <InfoRow label="Title" value={task.title} />
          <InfoRow label="Category" value={task.category.replace(/_/g, ' ')} />
          <InfoRow label="Priority" value={task.priority} />
          <InfoRow label="Status" value={task.status} />
          <InfoRow label="Assigned To" value={task.assignedTo} />
          <InfoRow label="Due Date" value={task.dueDate} />
          <InfoRow label="Description" value={task.description || 'No description provided.'} isLast />
        </AppCard>
 
        <View style={styles.actionSection}>
          {task.status === 'PENDING' && (
            <AppButton title="Start Task" onPress={handleStart} variant="primary" fullWidth />
          )}
          {task.status === 'IN_PROGRESS' && (
            <AppButton title="Complete Task" onPress={handleComplete} variant="success" fullWidth />
          )}
          {task.status === 'COMPLETED' && (
            <View style={styles.btnRow}>
              <AppButton 
                title="Reopen" 
                onPress={handleReopen} 
                variant="danger" 
                style={{ flex: 1, marginRight: Spacing.sm }} 
              />
              <AppButton 
                title="Verify" 
                onPress={handleVerify} 
                variant="success" 
                style={{ flex: 1 }} 
              />
            </View>
          )}
        </View>
 
        <ComplianceSafetyNotice />
      </ScrollView>
    </ScreenContainer>
  );
}
 
// 5. CreateComplianceTaskScreen
export function CreateComplianceTaskScreen({ navigation }: any) {
  const [title, setTitle] = useState('');
  const [category, setCategory] = useState('WASTE');
  const [priority, setPriority] = useState('MEDIUM');
  const [assignedTo, setAssignedTo] = useState('');
  const [dueDate, setDueDate] = useState('');
  const [description, setDescription] = useState('');
 
  const createMutation = useCreateComplianceTask();
 
  const categoryOptions = [
    { label: 'Waste', value: 'WASTE' },
    { label: 'Housekeeping', value: 'HOUSEKEEPING' },
    { label: 'Lift Safety', value: 'LIFT_SAFETY' },
    { label: 'Fire Safety', value: 'FIRE_SAFETY' },
  ];
 
  const priorityOptions = [
    { label: 'Low', value: 'LOW' },
    { label: 'Medium', value: 'MEDIUM' },
    { label: 'High', value: 'HIGH' },
    { label: 'Critical', value: 'CRITICAL' },
  ];
 
  const handleSubmit = async () => {
    if (!title || !assignedTo || !dueDate) {
      Alert.alert('Error', 'Please fill in all required fields.');
      return;
    }
    await createMutation.mutateAsync({
      title,
      category,
      priority,
      assignedTo,
      dueDate,
      description,
    });
    Alert.alert('Success', 'Compliance task created successfully.');
    navigation.goBack();
  };
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader title="New Compliance Task" onBack={() => navigation.goBack()} />
      <ScrollView contentContainerStyle={styles.scroll} showsVerticalScrollIndicator={false}>
        <AppCard padding="lg" style={{ marginBottom: Spacing.xl }}>
          <FormField 
            label="Task Title" 
            value={title} 
            onChangeText={setTitle} 
            placeholder="e.g., Conduct Fire Drill" 
            required 
          />
 
          <Text style={styles.formLabel}>Category *</Text>
          <FilterChips 
            options={categoryOptions} 
            selected={category} 
            onChange={setCategory} 
            style={{ marginBottom: Spacing.lg }}
          />
 
          <Text style={styles.formLabel}>Priority *</Text>
          <FilterChips 
            options={priorityOptions} 
            selected={priority} 
            onChange={setPriority} 
            style={{ marginBottom: Spacing.lg }}
          />
 
          <FormField 
            label="Assigned To" 
            value={assignedTo} 
            onChangeText={setAssignedTo} 
            placeholder="e.g., Ramesh Kumar" 
            required 
          />
 
          <FormField 
            label="Due Date" 
            value={dueDate} 
            onChangeText={setDueDate} 
            placeholder="e.g., 2026-07-15" 
            required 
          />
 
          <FormField 
            label="Description" 
            value={description} 
            onChangeText={setDescription} 
            placeholder="Enter additional details..." 
            multiline 
            numberOfLines={3} 
          />
 
          <AppButton 
            title={createMutation.isPending ? 'Creating...' : 'Create Task'} 
            onPress={handleSubmit} 
            variant="primary" 
            fullWidth 
            disabled={createMutation.isPending}
            style={{ marginTop: Spacing.md }}
          />
        </AppCard>
      </ScrollView>
    </ScreenContainer>
  );
}
