import React from 'react';
import { FlatList } from 'react-native';
import type { NativeStackScreenProps } from '@react-navigation/native-stack';
import { AppButton } from '../../../shared/components/AppButton';
import { AppCard } from '../../../shared/cards/AppCard';
import { EmptyState } from '../../../shared/feedback/EmptyState';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { FormField } from '../../../shared/forms/FormField';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import { StatusBadge, getParkingBadgeType } from '../../../shared/components/StatusBadge';
import type { FacilityOpsStackParamList } from '../../../app/navigation/navigation.types';
import { DetailRow, labelize, ParkingScreen } from '../../parking/components/ParkingUi';
import { useInventory } from '../data/useInventory';
 
type Props = NativeStackScreenProps<FacilityOpsStackParamList, 'InventoryList'>;
 
export function InventoryListScreen({ navigation }: Props) {
  const [query, setQuery] = React.useState('');
  const { data = [], isLoading, error, refetch } = useInventory({ query });
  if (isLoading) return <LoadingState message="Loading inventory..." showCardPlaceholder />;
  if (error) return <ErrorState message={error.message} onRetry={refetch} />;
  return (
    <ParkingScreen title="Inventory" subtitle="Issue and return tracking" onBack={navigation.goBack}>
      <FormField label="Search" value={query} onChangeText={setQuery} placeholder="Item, category, store room" />
      <FlatList data={data} keyExtractor={(item) => item.id} scrollEnabled={false} ListEmptyComponent={<EmptyState title="No inventory" description="No inventory items found." />} renderItem={({ item }) => (
        <AppCard style={{ marginBottom: 12 }}>
          <StatusBadge label={labelize(item.stockStatus)} type={getParkingBadgeType(item.stockStatus)} />
          <DetailRow label="Item" value={item.itemName} />
          <DetailRow label="Category" value={labelize(item.category)} />
          <DetailRow label="Current stock" value={`${item.currentStock} ${item.unit}`} />
          <DetailRow label="Minimum stock" value={`${item.minimumStockLevel} ${item.unit}`} />
          <DetailRow label="Location" value={item.location} />
          <DetailRow label="Last issued" value={item.lastIssuedDate} />
          <DetailRow label="Last purchased" value={item.lastPurchasedDate} />
          <AppButton title="Issue / Return" onPress={() => navigation.navigate('InventoryTransaction', { itemId: item.id })} variant="secondary" compact />
        </AppCard>
      )} />
      <AppButton title="Purchase Request Placeholder" onPress={() => navigation.navigate('PurchaseRequest')} variant="ghost" />
    </ParkingScreen>
  );
}
