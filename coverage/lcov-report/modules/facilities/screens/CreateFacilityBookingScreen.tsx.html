import React from 'react';
import type { NativeStackScreenProps } from '@react-navigation/native-stack';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { FormField } from '../../../shared/forms/FormField';
import type { FacilityStackParamList } from '../../../app/navigation/navigation.types';
import type { FacilityBookingType } from '../../../shared/types/facility.types';
import { FooterActions, ParkingScreen, Selector } from '../../parking/components/ParkingUi';
import { useCreateFacilityBooking } from '../data/useCreateFacilityBooking';
import { useFacilityDetail } from '../data/useFacilityDetail';
 
type Props = NativeStackScreenProps<FacilityStackParamList, 'CreateFacilityBooking'>;
 
export function CreateFacilityBookingScreen({ navigation, route }: Props) {
  const facilityId = route.params.facilityId ?? 'facility-1';
  const { data: facility } = useFacilityDetail(facilityId);
  const { submit, isSubmitting, error } = useCreateFacilityBooking();
  const [purpose, setPurpose] = React.useState('');
  const [date, setDate] = React.useState('2026-07-01');
  const [slot, setSlot] = React.useState(route.params.selectedSlotId ? 'Selected available slot' : '');
  const [guestCount, setGuestCount] = React.useState('');
  const [contact, setContact] = React.useState('');
  const [setupRequired, setSetupRequired] = React.useState<'NO' | 'YES'>('NO');
  const [equipment, setEquipment] = React.useState('');
  const [requirements, setRequirements] = React.useState('');
  const [rulesAccepted, setRulesAccepted] = React.useState<'NO' | 'YES'>('NO');
  const [damageAccepted, setDamageAccepted] = React.useState<'NO' | 'YES'>('NO');
  const [errors, setErrors] = React.useState<Record<string, string>>({});
 
  async function handleSubmit() {
    const next: Record<string, string> = {};
    const guests = Number(guestCount);
    if (!facility) next.facility = 'Facility is required.';
    if (!date.trim()) next.date = 'Date is required.';
    if (!slot.trim()) next.slot = 'Slot is required.';
    if (!purpose.trim()) next.purpose = 'Purpose is required.';
    if (!guestCount.trim() || guests <= 0) next.guestCount = 'Guest count must be greater than 0.';
    if (facility && guests > facility.capacity) next.guestCount = 'Guest count exceeds facility capacity.';
    if (!/^\d{10}$/.test(contact.trim())) next.contact = 'Contact number must be 10 digits.';
    if (rulesAccepted !== 'YES') next.rulesAccepted = 'Facility rules agreement is required.';
    if (damageAccepted !== 'YES') next.damageAccepted = 'Damage responsibility consent is required.';
    setErrors(next);
    if (Object.keys(next).length > 0 || !facility) return;
    const result = await submit({
      unitId: 'unit-a-1204',
      facilityId: facility.id,
      facilityName: facility.name,
      bookingType: facility.bookingType as FacilityBookingType,
      date,
      slot,
      purpose,
      guestCount: guests,
      contactNumber: contact,
      specialRequirements: requirements,
      setupRequired: setupRequired === 'YES',
      equipmentRequired: equipment,
      rulesAccepted: rulesAccepted === 'YES',
      damageConsentAccepted: damageAccepted === 'YES',
      chargeAmount: facility.chargeAmount,
      depositAmount: facility.depositAmount,
      approvalRequired: facility.approvalRequired,
    });
    if (result.ok) navigation.navigate('FacilityBookingReview', { bookingId: result.data.id });
  }
 
  return (
    <ParkingScreen title="Create Booking" subtitle={facility?.name ?? 'Facility'} onBack={navigation.goBack} footer={<FooterActions primaryLabel="Review booking" onPrimary={handleSubmit} loading={isSubmitting} />}>
      {error ? <ErrorState message={error.message} /> : null}
      <FormField label="Facility" value={facility?.name ?? ''} onChangeText={() => undefined} editable={false} error={errors.facility} />
      <FormField label="Date" required value={date} onChangeText={setDate} error={errors.date} />
      <FormField label="Selected slot" required value={slot} onChangeText={setSlot} error={errors.slot} />
      <FormField label="Booking purpose" required value={purpose} onChangeText={setPurpose} error={errors.purpose} />
      <FormField label="Number of guests" required value={guestCount} onChangeText={setGuestCount} keyboardType="number-pad" error={errors.guestCount} />
      <FormField label="Resident contact number" required value={contact} onChangeText={setContact} keyboardType="phone-pad" maxLength={10} error={errors.contact} />
      <FormField label="Special requirements" value={requirements} onChangeText={setRequirements} multiline numberOfLines={3} />
      <Selector label="Setup required" value={setupRequired} onChange={setSetupRequired} options={[{ label: 'No', value: 'NO' }, { label: 'Yes', value: 'YES' }]} />
      <FormField label="Equipment required optional" value={equipment} onChangeText={setEquipment} />
      <Selector label="Facility rules agreement" value={rulesAccepted} onChange={setRulesAccepted} error={errors.rulesAccepted} options={[{ label: 'I agree', value: 'YES' }, { label: 'Not yet', value: 'NO' }]} />
      <Selector label="Damage responsibility consent" value={damageAccepted} onChange={setDamageAccepted} error={errors.damageAccepted} options={[{ label: 'I accept', value: 'YES' }, { label: 'Not yet', value: 'NO' }]} />
    </ParkingScreen>
  );
}
