import React, { useState } from 'react';
import { Pressable, StyleSheet, View } from 'react-native';
import Animated, { FadeInDown, FadeInUp } from 'react-native-reanimated';
import { useAppTheme } from '../../../shared/theme/useAppTheme';
import { useExpectedVisitors } from '../data/useExpectedVisitors';
import { useGuardDashboard } from '../data/useGuardDashboard';
import { useStaffToday } from '../data/useStaffCheckIn';
import { ScreenScaffold } from '../../../shared/layout/ScreenScaffold';
import { Stack } from '../../../shared/layout/Stack';
import { Box } from '../../../shared/layout/Box';
import { Inline } from '../../../shared/layout/Inline';
import { ResponsiveGrid } from '../../../shared/layout/ResponsiveGrid';
import { SectionBlock } from '../../../shared/layout/SectionBlock';
import { AppText } from '../../../shared/components/AppText';
import { AppCard } from '../../../shared/cards/AppCard';
import { AppIcon } from '../../../shared/icons/AppIcon';
import { AppIconBubble } from '../../../shared/icons/AppIconBubble';
import { GradientCard } from '../../../shared/components/GradientCard';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import { ListItemCard } from '../../../shared/cards/ListItemCard';
import { QuickActionCard } from '../../../shared/cards/QuickActionCard';
import { StatusBadge } from '../../../shared/components/StatusBadge';
import type { GuardHomeScreenProps } from '../../../app/navigation/navigation.types';
 
export function GuardHomeScreen({ navigation }: GuardHomeScreenProps) {
  const { colors, radius, shadows } = useAppTheme();
  const [isOnline, setIsOnline] = useState(true);
  const { data: dashboard, isLoading, error, refetch } = useGuardDashboard();
  const { data: expectedVisitors } = useExpectedVisitors();
  const { data: staffMembers } = useStaffToday();
 
  if (isLoading) {
    return (
      <ScreenScaffold style={styles.container}>
        <LoadingState message="Loading gate console..." />
      </ScreenScaffold>
    );
  }
 
  if (error || !dashboard?.profile) {
    return (
      <ScreenScaffold style={styles.container}>
        <ErrorState message={error?.message || "Guard dashboard data is unavailable."} onRetry={refetch} />
      </ScreenScaffold>
    );
  }
 
  const { profile, activityLogs = [], offlinePendingCount = 0 } = dashboard;
  const expectedCount = expectedVisitors?.length ?? 0;
  const staffCheckedIn = staffMembers?.filter((s) => s.attendanceStatus === 'CHECKED_IN').length ?? 0;
 
  const handleQuickAction = (route: any) => {
    navigation.navigate(route);
  };
 
  const handleTabAction = (route: any) => {
    navigation.navigate(route);
  };
 
  return (
    <ScreenScaffold scroll edges={['top', 'bottom']} style={styles.container}>
      <Stack gap="lg">
        {/* ─── Operational Hero Card ─── */}
        <Animated.View entering={FadeInUp.duration(400)}>
          <GradientCard variant="guard" style={styles.heroCard}>
            <Inline justify="space-between" align="center" wrap={false}>
              <Box flex={1}>
                <AppText variant="screenSubtitle" style={styles.heroSubtitle}>
                  GATE CONTROL CENTER
                </AppText>
                <AppText variant="screenTitle" style={styles.heroTitle}>
                  {profile.name}
                </AppText>
                <Inline gap="xs" style={{ marginTop: 8 }}>
                  <Box
                    px="sm"
                    py="xxs"
                    backgroundColor="rgba(255,255,255,0.18)"
                    borderRadius={radius.pill}
                  >
                    <AppText variant="caption" style={styles.heroChipText}>
                      {profile.gateName}
                    </AppText>
                  </Box>
                  <AppText variant="caption" style={styles.heroSubText}>
                    Shift: {profile.shiftName}
                  </AppText>
                </Inline>
              </Box>
 
              <Pressable
                onPress={() => setIsOnline(!isOnline)}
                style={[
                  styles.statusToggle,
                  isOnline
                    ? { backgroundColor: colors.successSoft, borderColor: colors.success }
                    : { backgroundColor: colors.surfaceSoft, borderColor: colors.disabled },
                ]}
              >
                <View
                  style={[
                    styles.statusDot,
                    { backgroundColor: isOnline ? colors.success : colors.textMuted },
                  ]}
                />
                <AppText
                  variant="caption"
                  style={[
                    styles.statusToggleText,
                    { color: isOnline ? colors.success : colors.textSecondary },
                  ]}
                >
                  {isOnline ? 'ONLINE' : 'OFFLINE'}
                </AppText>
              </Pressable>
            </Inline>
          </GradientCard>
        </Animated.View>
 
        {/* ─── Offline Queue / Sync Sync ─── */}
        {offlinePendingCount > 0 && (
          <Animated.View entering={FadeInUp.delay(100).duration(400)}>
            <AppCard
              variant="warning"
              style={styles.bannerCard}
              onPress={() => handleQuickAction('OfflineQueue')}
              pressable
            >
              <Inline gap="sm" align="center" justify="space-between" wrap={false}>
                <Inline gap="xs" align="center" style={{ flex: 1 }}>
                  <AppIcon name="warning" size={20} color={colors.warning} />
                  <AppText variant="bodySmall" tone="warning" style={{ fontWeight: '600', flex: 1 }}>
                    {offlinePendingCount} entries waiting to sync offline logs.
                  </AppText>
                </Inline>
                <AppIcon name="chevronRight" size={16} color={colors.warning} />
              </Inline>
            </AppCard>
          </Animated.View>
        )}
 
        {/* ─── Large Verify Triggers ─── */}
        <Animated.View entering={FadeInUp.delay(150).duration(400)}>
          <ResponsiveGrid columnsPhone={2} columnsTablet={2} gap={16}>
            <AppCard
              pressable
              onPress={() => handleQuickAction('PassSearch')}
              style={[styles.primaryActionCard, { borderColor: colors.primary }]}
            >
              <Box alignItems="center" py="md">
                <AppIconBubble name="qr" size={56} iconSize={26} color={colors.primary} backgroundColor={colors.primarySoft} />
                <AppText variant="sectionTitle" style={styles.primaryActionTitle}>
                  Scan / Verify Pass
                </AppText>
                <AppText variant="caption" tone="secondary" style={styles.primaryActionSub}>
                  Scan QR code or search code
                </AppText>
              </Box>
            </AppCard>
 
            <AppCard
              pressable
              onPress={() => handleQuickAction('ManualEntry')}
              style={[styles.primaryActionCard, { borderColor: colors.guard }]}
            >
              <Box alignItems="center" py="md">
                <AppIconBubble name="noc" size={56} iconSize={26} color={colors.guard || '#0F766E'} backgroundColor={`${colors.guard || '#0F766E'}15`} />
                <AppText variant="sectionTitle" style={styles.primaryActionTitle}>
                  New Walk-in
                </AppText>
                <AppText variant="caption" tone="secondary" style={styles.primaryActionSub}>
                  Record a guest manually
                </AppText>
              </Box>
            </AppCard>
          </ResponsiveGrid>
        </Animated.View>
 
        {/* ─── Operations Grid ─── */}
        <Animated.View entering={FadeInUp.delay(200).duration(400)}>
          <SectionBlock title="Gate Entry Options">
            <ResponsiveGrid columnsPhone={2} columnsTablet={4} gap={12}>
              <QuickActionCard
                title="Quick Entry"
                subtitle="Cabs / Deliveries"
                iconName="vehicle"
                onPress={() => handleQuickAction('QuickEntry')}
              />
              <QuickActionCard
                title="Staff Entry"
                subtitle="Attendance log"
                iconName="staff"
                onPress={() => handleQuickAction('StaffCheckIn')}
              />
              <QuickActionCard
                title="Vehicle Lookup"
                subtitle="Search vehicle no."
                iconName="search"
                onPress={() => handleQuickAction('GuardVehicleLookup')}
              />
              <QuickActionCard
                title="SOS Alert"
                subtitle="Incident report"
                iconName="warning"
                variant="danger"
                onPress={() => handleQuickAction('EmergencyAlert')}
              />
            </ResponsiveGrid>
          </SectionBlock>
        </Animated.View>
 
        {/* ─── Operational Metrics ─── */}
        <Animated.View entering={FadeInUp.delay(250).duration(400)}>
          <SectionBlock title="Shift Log Metrics">
            <ResponsiveGrid columnsPhone={2} columnsTablet={2} gap={12}>
              <AppCard
                pressable
                onPress={() => handleTabAction('GuardExpectedTab')}
                style={styles.metricRowCard}
              >
                <Inline justify="space-between" align="center" wrap={false}>
                  <Box>
                    <AppText variant="caption" tone="secondary">
                      Expected Today
                    </AppText>
                    <AppText variant="largeAmount" style={{ color: colors.primary, fontWeight: '800' }}>
                      {expectedCount}
                    </AppText>
                  </Box>
                  <AppIconBubble name="visitor" size={44} iconSize={20} />
                </Inline>
              </AppCard>
 
              <AppCard
                pressable
                onPress={() => handleQuickAction('StaffCheckIn')}
                style={styles.metricRowCard}
              >
                <Inline justify="space-between" align="center" wrap={false}>
                  <Box>
                    <AppText variant="caption" tone="secondary">
                      Staff Checked In
                    </AppText>
                    <AppText variant="largeAmount" style={{ color: colors.success, fontWeight: '800' }}>
                      {staffCheckedIn}
                    </AppText>
                  </Box>
                  <AppIconBubble name="staff" size={44} iconSize={20} color={colors.success} backgroundColor={colors.successSoft} />
                </Inline>
              </AppCard>
            </ResponsiveGrid>
          </SectionBlock>
        </Animated.View>
 
        {/* ─── Recent Logs ─── */}
        <Animated.View entering={FadeInDown.delay(300).duration(400)}>
          <SectionBlock
            title="Recent Gate Activity"
            action={
              <Pressable onPress={() => handleTabAction('GuardLogsTab')}>
                <AppText variant="caption" style={{ color: colors.primary, fontWeight: '600' }}>
                  View Logs
                </AppText>
              </Pressable>
            }
          >
            {activityLogs.length > 0 ? (
              activityLogs.slice(0, 3).map((log) => {
                const isCheckIn = log.status === 'CHECKED_IN';
                return (
                  <ListItemCard
                    key={log.id}
                    title={log.personName}
                    subtitle={`Flat ${log.flatOrCommon} · via ${log.approvalSource}`}
                    meta={log.time}
                    iconName={log.activityType === 'DELIVERY' ? 'visitor' : log.activityType === 'CAB' ? 'vehicle' : 'person'}
                    right={
                      <Box px="sm" py="xxs" borderRadius={4} backgroundColor={isCheckIn ? colors.successSoft : colors.dangerSoft}>
                        <AppText variant="caption" style={{ fontWeight: '700', color: isCheckIn ? colors.success : colors.danger }}>
                          {isCheckIn ? 'IN' : 'OUT'}
                        </AppText>
                      </Box>
                    }
                  />
                );
              })
            ) : (
              <AppCard>
                <AppText variant="bodySmall" tone="muted" style={{ textAlign: 'center' }}>
                  No gate records logged in this shift.
                </AppText>
              </AppCard>
            )}
          </SectionBlock>
        </Animated.View>
      </Stack>
    </ScreenScaffold>
  );
}

    flex: 1,
