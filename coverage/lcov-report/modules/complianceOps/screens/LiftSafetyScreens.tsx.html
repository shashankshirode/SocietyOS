// 4. LiftBreakdownReportScreen
export function LiftBreakdownReportScreen({ route, navigation }: any) {
  const { liftId } = route.params;
  const [breakdownType, setBreakdownType] = useState('TECHNICAL_GLITCH');
  const [description, setDescription] = useState('');
 
  const reportMutation = useReportLiftBreakdown();
 
  const handleSubmit = async () => {
    if (!description) {
      Alert.alert('Error', 'Please enter description of the issue.');
      return;
    }
    await reportMutation.mutateAsync({
      liftId,
      input: { breakdownType, description },
    });
    Alert.alert('Success', 'Breakdown reported. AMC vendor has been alerted.');
    navigation.pop();
  };
 
  const typeOptions = [
    { label: 'Technical Glitch', value: 'TECHNICAL_GLITCH' },
    { label: 'Power Failure', value: 'POWER_FAILURE' },
    { label: 'Physical Damage', value: 'PHYSICAL_DAMAGE' },
    { label: 'Other', value: 'OTHER' },
  ];
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader title="Report Lift Breakdown" onBack={() => navigation.goBack()} />
      <ScrollView contentContainerStyle={styles.scroll} showsVerticalScrollIndicator={false}>
        <AppCard padding="lg" style={{ marginBottom: Spacing.xl }}>
          <Text style={styles.formLabel}>Breakdown Type *</Text>
          <FilterChips 
            options={typeOptions} 
            selected={breakdownType} 
            onChange={setBreakdownType} 
            style={{ marginBottom: Spacing.lg }}
          />
 
          <FormField 
            label="Issue Description" 
            value={description} 
            onChangeText={setDescription} 
            placeholder="e.g., Lift stuck on 4th floor, doors not opening..." 
            multiline 
            numberOfLines={3} 
            required
          />
 
          <AppButton 
            title={reportMutation.isPending ? 'Reporting...' : 'Report Breakdown'} 
            onPress={handleSubmit} 
            variant="danger" 
            fullWidth 
            disabled={reportMutation.isPending}
            style={{ marginTop: Spacing.md }}
          />
        </AppCard>
      </ScrollView>
    </ScreenContainer>
  );
}
 
// 5. LiftMaintenanceVisitLogScreen
export function LiftMaintenanceVisitLogScreen({ route, navigation }: any) {
  const { liftId } = route.params;
  const { data, refetch } = useLiftMaintenanceVisitLog(liftId);
  const addMutation = useAddLiftMaintenanceVisit();
  const list = (data as any[]) || [];
 
  const handleAddVisit = () => {
    Alert.alert(
      'Log Routine Maintenance',
      'Log a routine preventative maintenance visit for this elevator?',
      [
        { text: 'Cancel', style: 'cancel' },
        { 
          text: 'Log Visit', 
          onPress: async () => {
            await addMutation.mutateAsync({
              liftId,
              input: { visitType: 'ROUTINE', technicianName: 'AMC Technician', findings: 'All safety parameters verified.' }
            });
            Alert.alert('Success', 'Maintenance visit logged.');
            refetch();
          } 
        }
      ]
    );
  };
 
  return (
    <ScreenContainer edges={['top', 'bottom']}>
      <ResponsivePageHeader 
        title="Maintenance Visits" 
        subtitle="Elevator service history" 
        onBack={() => navigation.goBack()} 
        rightActions={
          <AppButton
            variant="primary"
            size="sm"
            title="Log Visit"
            onPress={handleAddVisit}
          />
        }
      />
      <FlatList
        data={list}
        keyExtractor={(item) => item.id}
        contentContainerStyle={styles.list}
        showsVerticalScrollIndicator={false}
        renderItem={({ item }) => (
          <AppCard style={styles.hkCard} padding="md">
            <InfoRow label="Date" value={item.visitDate} />
            <InfoRow label="Visit Type" value={item.visitType} />
            <InfoRow label="Technician" value={item.technicianName} />
            <InfoRow label="Status" value={item.status} />
            <InfoRow label="Notes" value={item.findings || 'No findings recorded.'} isLast />
          </AppCard>
        )}
      />
    </ScreenContainer>
  );
}
