import React from 'react';
import { LoadingState } from '../../../shared/feedback/LoadingState';
import { ErrorState } from '../../../shared/feedback/ErrorState';
import { StatusBadge, getParkingBadgeType } from '../../../shared/components/StatusBadge';
import { useParkingHardwareReadiness } from '../data/useParkingHardwareReadiness';
import { DetailCard, DetailRow, labelize, ParkingScreen, WarningText } from '../components/ParkingUi';
import type { ParkingAccessReadinessScreenProps } from '../../../app/navigation/navigation.types';
 
export function ParkingAccessReadinessScreen({ navigation, route }: ParkingAccessReadinessScreenProps) {
  const { data, isLoading, error, refetch } = useParkingHardwareReadiness(route.params.societyId);
 
  if (isLoading) {
    return <LoadingState message="Loading access readiness..." showCardPlaceholder />;
  }
 
  if (error) {
    return <ErrorState message={error.message} onRetry={refetch} />;
  }
 
  if (!data) {
    return <ErrorState message="Hardware readiness is unavailable." onRetry={refetch} />;
  }
 
  return (
    <ParkingScreen title="Access Readiness" subtitle="RFID, ANPR, barriers, EV, and sensors" onBack={navigation.goBack}>
      <WarningText>Hardware integrations will be connected later through society-approved devices and backend integration APIs.</WarningText>
      <DetailCard title="Readiness checklist">
        <StatusBadge label={labelize(data.anprReadiness)} type={getParkingBadgeType(data.anprReadiness)} />
        <DetailRow label="RFID readiness" value={labelize(data.rfidReadiness)} />
        <DetailRow label="ANPR readiness" value={labelize(data.anprReadiness)} />
        <DetailRow label="Boom barrier readiness" value={labelize(data.boomBarrierReadiness)} />
        <DetailRow label="EV charging readiness" value={labelize(data.evChargingReadiness)} />
        <DetailRow label="Smart parking sensor readiness" value={labelize(data.smartSensorReadiness)} />
        <DetailRow label="Notes" value={data.notes} isLast />
      </DetailCard>
    </ParkingScreen>
  );
}
